# Build type: Release unless one is given, e.g., with -DCMAKE_BUILD_TYPE=Debug;
# a build with no type has no optimization at all. Multi-config generators
# (Visual Studio, Xcode) choose it at build time and are left alone

    set(CMAKE_BUILD_TYPE "Release" CACHE STRING "Build type" FORCE)
endif ()

# Add compilation and linking flags
# Note: add_compile_options() is recommended over CMAKE_<LANG>_FLAGS_<CONFIG>
# Each option is given to the compilers and the linkers that take it, which
# is checked once, rather than chosen by the name of the compiler; those
# checked with the C++ compiler are given to the C++ sources
include(CheckCXXCompilerFlag)
include(CheckLinkerFlag)

# CPLEX is linked by its functions only, whether static or not
add_compile_definitions(BUILD_CPXSTATIC CPXSIZE_BITS_TEST_DISABLE)

# NOMINMAX prevents <windows.h> from defining the min()/max() function-like
# macros, which otherwise break every `std::min`/`std::max` call (FastFlow
# uses them in ff/parallel_for_internals.hpp). FastFlow does define NOMINMAX
# itself, but only inside ff/platforms/platform.h, i.e. too late if any other
# header pulled in <windows.h> first, so we define it globally and up front;
# it means nothing to any other header.
add_compile_definitions(NOMINMAX)

# The warnings silenced where the compiler takes -W (e.g., GCC and Clang),
# each where the compiler knows it, i.e., takes the option that enables it
foreach (_w enum-compare enum-conversion deprecated-declarations
        undefined-internal deprecated-enum-enum-conversion
        deprecated-enum-float-conversion free-nonheap-object)
    string(MAKE_C_IDENTIFIER "SMSpp_CXX_W_${_w}" _var)
    check_cxx_compiler_flag("-W${_w}" ${_var})
    if (${_var})
        add_compile_options("-Wno-${_w}")
    endif ()
endforeach ()
unset(_w)
unset(_var)

# /MP compiles the sources of a target in parallel; /Zc:__cplusplus forces
# MSVC to report the real value of the __cplusplus macro (by default it is
# stuck at 199711L even in /std:c++17 or /std:c++20 mode), and FastFlow
# guards all its C++11 code paths with `#if __cplusplus >= 201103L`, so
# without this option those paths are skipped and the build fails in
# ff/spin-lock.hpp, ff/utils.hpp, ff/dynqueue.hpp, ... (other compilers
# report __cplusplus correctly and do not take the option)
foreach (_opt "/MP" "/Zc:__cplusplus")
    string(MAKE_C_IDENTIFIER "SMSpp_CXX_${_opt}" _var)
    check_cxx_compiler_flag("${_opt}" ${_var})
    if (${_var})
        add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:${_opt}>")
    endif ()
endforeach ()
unset(_opt)
unset(_var)
