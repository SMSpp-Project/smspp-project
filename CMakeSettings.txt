if (MSVC)
    set(MSVC_DISABLED_WARNINGS_LIST
        "C4018" # 'expression' : signed/unsigned mismatch
        "C4057" # 'operator' : 'identifier1' indirection to
                # slightly different base types from 'identifier2'
        "C4060" # switch statement contains no 'case' or 'default' labels
        "C4065" # switch statement contains 'default' but no 'case' labels
        "C4068" # unknown pragma 'identifier'
        "C4100" # 'identifier' : unreferenced formal parameter
        "C4101" # 'identifier' : unreferenced local variable
        "C4102" # 'label' : unreferenced label
        "C4127" # conditional expression is constant
        "C4146" # unary minus operator applied to unsigned type,
                # result still unsigned
        "C4189" # 'identifier' : local variable is initialized but not referenced
        "C4244" # 'argument' : conversion from 'type1' to 'type2',
                # possible loss of data
        "C4245" # 'conversion' : conversion from 'type1' to 'type2',
                # signed/unsigned mismatch
        "C4267" # 'var' : conversion from 'size_t' to 'type',
                # possible loss of data
        "C4373" # 'function' : virtual function overrides 'base_function',
                # previous versions of the compiler did not override when
                # parameters only differed by const/volatile qualifiers
        "C4389" # 'operator' : signed/unsigned mismatch
        "C4390" # ';' : empty controlled statement found; is this the intent?
        "C4456" # declaration of 'identifier' hides previous local declaration
        "C4457" # declaration of 'identifier' hides function parameter
        "C4458" # declaration of 'identifier' hides class member
        "C4459" # declaration of 'identifier' hides global declaration
        "C4505" # 'function' : unreferenced function with internal linkage
                # has been removed
        "C4701" # potentially uninitialized local variable 'name' used
        "C4702" # unreachable code
        "C4703" # potentially uninitialized local pointer variable 'name' used
        "C4706" # assignment within conditional expression
        "C4717" # 'function' : recursive on all control paths, function will
                # cause runtime stack overflow
        "C4804" # 'operation' : unsafe use of type 'bool' in operation
        "C4834" # discarding return value of function with 'nodiscard' attribute
        "C4858" # discarding return value: this call discards the return value
                # of a function whose return type is marked [[nodiscard]]
        "C4996" # The POSIX name for this item is deprecated.
                # Instead, use the ISO C and C++ conformant name
        "C5054" # operator 'operator' : deprecated between enumerations of
                # different types (C++20)
    )
    string(REPLACE "C" " -wd" MSVC_DISABLED_WARNINGS_STR ${MSVC_DISABLED_WARNINGS_LIST})
    string(REGEX REPLACE "[/-]W[1234][ ]?" "" CMAKE_CXX_FLAGS ${CMAKE_CXX_FLAGS})
    # /Zc:__cplusplus forces MSVC to report the real value of the __cplusplus
    # macro (by default it is stuck at 199711L even in /std:c++17 or /std:c++20
    # mode). FastFlow guards all its C++11 code paths with `#if __cplusplus >=
    # 201103L`, so without this flag those paths are skipped and the build fails
    # in ff/spin-lock.hpp, ff/utils.hpp, ff/dynqueue.hpp, ... (works on Linux
    # because GCC/Clang already report __cplusplus correctly).
    # NOMINMAX prevents <windows.h> from defining the min()/max() function-like
    # macros, which otherwise break every `std::min`/`std::max` call (FastFlow
    # uses them in ff/parallel_for_internals.hpp). FastFlow does define NOMINMAX
    # itself, but only inside ff/platforms/platform.h, i.e. too late if any other
    # header pulled in <windows.h> first, so we define it globally and up front.
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -MP -W4 /Zc:__cplusplus /DNOMINMAX ${MSVC_DISABLED_WARNINGS_STR}")
    add_link_options("/IGNORE:4006,4075,4088,4255")
else () # Unix
    add_compile_options(
             "-Wno-enum-compare" "-Wno-enum-conversion" "-Wno-deprecated-declarations" "-Wno-undefined-internal"
             "-Wno-deprecated-enum-enum-conversion" "-Wno-deprecated-enum-float-conversion" "-Wno-free-nonheap-object" # C++20
             "-DBUILD_CPXSTATIC" "-DCPXSIZE_BITS_TEST_DISABLE"
             #"-ferror-limit=1" "-fexceptions"
             #"-Wall" "-Wpedantic" "-Wextra"
             #"-fsanitize=undefined" "-fsanitize=address" "-fno-omit-frame-pointer"
             "$<$<CONFIG:DEBUG>:-fno-inline>"
             #"$<$<CONFIG:MINSIZEREL>:>"
             #"$<$<CONFIG:RELEASE>:>"
             #"$<$<CONFIG:RELWITHDEBINFO>:>"
    )
    #add_link_options("-fsanitize=address")
endif ()
