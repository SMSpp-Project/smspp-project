289  virtual RHSValue rel_viol( void ) const {
290  c_RHSValue rhs = get_rhs();
291  c_RHSValue val = value();
292 
293  // if the value is +INF, then if the RHS is < +INF then the constraint is
294  // infinitely violated, otherwise is infinitely slackened
295  if( val >= Inf<RHSValue>() )
296  return( rhs < Inf<RHSValue>() ? Inf<RHSValue>() : - Inf<RHSValue>() );
297 
298  c_RHSValue lhs = get_lhs();
299  // if the value is -INF, then if the LHS is > -INF then the constraint is
300  // infinitely violated, otherwise is infinitely slackened
301  if( val <= -Inf<RHSValue>() )
302  return( lhs > -Inf<RHSValue>() ? Inf<RHSValue>() : - Inf<RHSValue>() );
303 
304  // the value is finite
305  if( lhs <= - Inf<RHSValue>() )
306  if( rhs >= Inf<RHSValue>() )
307  return( -Inf<RHSValue>() );
308  else
309  return( rhs == 0 ? val : ( val - rhs ) / std::abs( rhs ) );
310  else
311  if( rhs >= Inf<RHSValue>() )
312  return( lhs == 0 ? - val : ( lhs - val ) / std::abs( lhs ) );
313 
314  // both LHS and RHS are finite
315  if( lhs == 0 )
316  if( rhs == 0 )
317  return( std::abs( val ) );
318  else
319  return( std::max( - val , val - rhs ) / std::abs( rhs ) );
320  else
321  if( rhs == 0 )
322  return( std::max( lhs - val , val ) / std::abs( lhs ) );
323  else
324  return( std::max( ( lhs - val ) / std::abs( lhs ) ,
325  ( val - rhs ) / std::abs( rhs ) ) );
326  }
