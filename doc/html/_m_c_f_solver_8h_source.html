608 template< class MCFC >
609 void MCFSolver< MCFC >::process_outstanding_Modification( void )
610 {
611  // no-frills loop: do them in order, with no attempt at optimizing
612  while( ! v_mod.empty() ) {
613  auto mod = v_mod.front(); // pick (a reference to) the first Modification
614 
615  /* Use a Lambda to define a "guts" of the method that can be called
616  recursively. Note the trick of defining the std::function object and
617  "passing" it to the lambda, which allows recursive calls. Note the need
618  to explicitly capture "this" to use fields/methods of the class. */
619 
620  auto MCFB = static_cast< MCFBlock * >( f_Block );
621 
622  std::function< void( sp_Mod )> guts_of_poM;
623  guts_of_poM = [ this , & guts_of_poM , MCFB ]( sp_Mod mod ) {
624  // process Modification - - - - - - - - - - - - - - - - - - - - - - - - - -
625  //- - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
626  /* This requires to patiently sift through the possible Modification types
627  to find what this Modification exactly is, and call the appropriate
628  method of MCFClass. */
629 
630  // GroupModification- - - - - - - - - - - - - - - - - - - - - - - - - - - -
631  {
632  const auto tmod = std::dynamic_pointer_cast<GroupModification>( mod );
633  if( tmod ) {
634  for( auto submod : tmod->v_sub_Modifications )
635  guts_of_poM( submod );
636 
637  return;
638  }
639  }
640 
641  // MCFBlockRngdMod- - - - - - - - - - - - - - - - - - - - - - - - - - - - -
642  /* Note: in the following we can assume that C, B and U are nonempty. This
643  is because they can be empty only if they are so when the object is
644  loaded. But if a Modification has been issued they are no longer empty
645  (a Modification changin nothing from the "empty" state is not issued).
646  */
647  {
648  const auto tmod = std::dynamic_pointer_cast<MCFBlockRngdMod>( mod );
649  if( tmod ) {
650  switch( tmod->f_type ) {
651  case( MCFBlockMod::eChgCost ):
652  if( tmod->f_stop == tmod->f_strt + 1 )
653  MCFC::ChgCost( tmod->f_strt , MCFB->get_C( tmod->f_strt ) );
654  else
655  MCFC::ChgCosts( MCFB->get_C().data() + tmod->f_strt , nullptr,
656  tmod->f_strt , tmod->f_stop );
657  break;
658 
659  case( MCFBlockMod::eChgCaps ):
660  if( tmod->f_stop == tmod->f_strt + 1 )
661  MCFC::ChgUCap( tmod->f_strt , MCFB->get_U( tmod->f_strt ) );
662  else
663  MCFC::ChgUCaps( MCFB->get_U().data() + tmod->f_strt , nullptr,
664  tmod->f_strt , tmod->f_stop );
665  break;
666 
667  case( MCFBlockMod::eChgDfct ):
668  if( tmod->f_stop == tmod->f_strt + 1 )
669  MCFC::ChgDfct( tmod->f_strt , MCFB->get_B( tmod->f_strt ) );
670  else
671  MCFC::ChgDfcts( MCFB->get_B().data() + tmod->f_strt , nullptr,
672  tmod->f_strt , tmod->f_stop );
673  break;
674 
675  case( MCFBlockMod::eOpenArc ):
676  for( auto arc = tmod->f_strt ; arc < tmod->f_stop ; )
677  MCFC::OpenArc( arc++ );
678  break;
679 
680  case( MCFBlockMod::eCloseArc ):
681  for( auto arc = tmod->f_strt ; arc < tmod->f_stop ; )
682  MCFC::CloseArc( arc++ );
683  break;
684 
685  default:
686  throw( std::invalid_argument( "unknown MCFBlockRngdMod type" ) );
687  }
688 
689  return;
690  }
691  }
692 
693  // MCFBlockSbstMod- - - - - - - - - - - - - - - - - - - - - - - - - - - - -
694  {
695  const auto tmod = std::dynamic_pointer_cast<MCFBlockSbstMod>( mod );
696  if( tmod ) {
697  switch( tmod->f_type ) {
698  case( MCFBlockMod::eOpenArc ):
699  for( auto arc : tmod->f_nms )
700  MCFC::OpenArc( arc );
701  return;
702 
703  case( MCFBlockMod::eCloseArc ):
704  for( auto arc : tmod->f_nms )
705  MCFC::CloseArc( arc );
706  return;
707  }
708 
709  // have to InINF-terminate the vector of indices (damn!)
710  MCFBlock::Vec_Index nmsI( tmod->f_nms.size() + 1 );
711  *copy( tmod->f_nms.begin() , tmod->f_nms.end() , nmsI.begin() ) =
712  Inf<MCFBlock::Index>();
713  switch( tmod->f_type ) {
714  case( MCFBlockMod::eChgCost ): {
715  MCFBlock::Vec_CNumber NCost( tmod->f_nms.size() );
716  auto C = MCFB->get_C();
717  for( MCFBlock::Index i = 0 ; i < NCost.size() ; i++ )
718  NCost[ i ] = C[ nmsI[ i ] ];
719 
720  MCFC::ChgCosts( NCost.data() , nmsI.data() );
721  break;
722  }
723 
724  case( MCFBlockMod::eChgCaps ): {
725  MCFBlock::Vec_FNumber NCap( tmod->f_nms.size() );
726  auto U = MCFB->get_U();
727  for( MCFBlock::Index i = 0 ; i < NCap.size() ; i++ )
728  NCap[ i ] = U[ nmsI[ i ] ];
729 
730  MCFC::ChgUCaps( NCap.data() , nmsI.data() );
731  break;
732  }
733 
734  case( MCFBlockMod::eChgDfct ): {
735  MCFBlock::Vec_FNumber NDfct( tmod->f_nms.size() );
736  auto B = MCFB->get_B();
737  for( MCFBlock::Index i = 0 ; i < NDfct.size() ; i++ )
738  NDfct[ i ] = B[ nmsI[ i ] ];
739 
740  MCFC::ChgDfcts( NDfct.data() , nmsI.data() );
741  break;
742  }
743 
744  default:
745  throw( std::invalid_argument( "unknown MCFBlockSbstMod type" ) );
746  }
747 
748  return;
749  }
750  }
751 
752  // MCFBlockMod- - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
753  // note: this is checked after the previous two because they derive from
754  // MCFBlockMod, and hence the std::dynamic_pointer_cast<> would suceed
755  {
756  const auto tmod = std::dynamic_pointer_cast<MCFBlockMod>( mod );
757  if( tmod ) {
758  // this is the "nuclear option": the MCFBlock has been re-loaded
759  MCFC::LoadNet( MCFB->get_NNodes() , MCFB->get_NArcs() ,
760  MCFB->get_NNodes() , MCFB->get_NArcs() ,
761  MCFB->get_U().size() ? MCFB->get_U().data() : nullptr ,
762  MCFB->get_C().size() ? MCFB->get_C().data() : nullptr ,
763  MCFB->get_B().size() ? MCFB->get_B().data() : nullptr ,
764  MCFB->get_SN().data() , MCFB->get_EN().data() );
765  MCFC::PreProcess();
766  return;
767  }
768  }
769 
770  // IMPORTANT NOTE: any remaining Modification is plainly ignored. It must
771  // be an "abstract" Modification, which this Solver does not need to look
772  // at
773 
774  }; // end( guts_of_poM ) - - - - - - - - - - - - - - - - - - - - - - - - -
775  //- - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
776 
777  // finally, call the "guts of" - - - - - - - - - - - - - - - - - - - - - - -
778 
779  guts_of_poM( mod ); // now the actual call
780 
781  v_mod.pop_front(); // now the Modification is processed: remove it
782 
783  } // end( while( there are Modification ) )
784  } // end( MCFSolver::process_outstanding_Modification )
