299  virtual RHSValue rel_viol( void ) const override {
300  RHSValue uval = f_function->get_upper_estimate();
301 
302  // if the upper estimate is +INF, then if the RHS is < +INF then the
303  // constraint is infinitely violated, otherwise is infinitely slackened
304  if( uval >= Inf<double>() )
305  return( f_rhs < Inf<double>() ? Inf<double>() : - Inf<double>() );
306 
307  RHSValue lval = f_function->get_lower_estimate();
308  // if the lower estimate is -INF, then if the LHS is > -INF then the
309  // constraint is infinitely violated, otherwise is infinitely slackened
310  if( lval <= -Inf<double>() )
311  return( f_lhs > -Inf<double>() ? Inf<double>() : - Inf<double>() );
312 
313  // both upper and lower estimate are finite
314  if( f_lhs <= - Inf<double>() )
315  if( f_rhs >= Inf<double>() )
316  return( -Inf<double>() );
317  else
318  return( f_rhs == 0 ? uval - f_rhs :
319  ( uval - f_rhs ) / std::abs( f_rhs ) );
320  else
321  if( f_rhs >= Inf<double>() )
322  return( f_lhs == 0 ? f_lhs - lval :
323  ( f_lhs - lval ) / std::abs( f_lhs ) );
324 
325  // both LHS and RHS are finite
326  if( f_lhs == 0 )
327  if( f_rhs == 0 )
328  return( std::max( f_lhs - lval , uval - f_rhs ) );
329  else
330  return( std::max( f_lhs - lval , uval - f_rhs ) / std::abs( f_rhs ) );
331  else
332  if( f_rhs == 0 )
333  return( std::max( f_lhs - lval , uval - f_rhs ) / std::abs( f_lhs ) );
334  else
335  return( std::max( ( f_lhs - lval ) / std::abs( f_lhs ) ,
336  ( uval - f_rhs ) / std::abs( f_rhs ) ) );
337  }
