477 inline void MCFCplex::SetPar( int par , int val )
478 {
479  try {
480  MCFClass::SetPar( par , val ); // is it handled by the base class?
481 
482  if( par == kMaxIter ) // let MaxIter to be handled by Cplex
483  CPXsetintparam( env , CPX_PARAM_NETITLIM , val > 0 ? val : 2100000000 );
484  }
485  catch( MCFException &e ) { // it is *not* handled by the base class
486  if( par == kQPMethod )
487  QPMthd = (QPMethod) val;
488  else
489  CPXsetintparam( env , par , val );
490  }
491  }
